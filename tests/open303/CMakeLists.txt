cmake_minimum_required(VERSION 3.23)
project(Open303Checks LANGUAGES CXX)
enable_testing()
add_subdirectory(../../lib/open303 open303)
add_executable(open303_checks engine_tests.cpp)
target_link_libraries(open303_checks PRIVATE open303)
add_test(NAME open303_checks COMMAND open303_checks)
option(OPEN303_SANITIZERS "Check the desktop DSP for memory and undefined-behavior errors" OFF)
if(OPEN303_SANITIZERS)
    foreach(target open303 open303_checks)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(open303_checks PRIVATE -fsanitize=address,undefined)
endif()
